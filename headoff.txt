1. حالة المشروع الحالية
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المشروع هو تطبيق إدارة مواقع بالذكاء الاصطناعي يعمل بالكامل من الموبايل.

البنية التقنية:
- الاستضافة: Cloudflare Pages (Frontend + API)
- قاعدة البيانات: Cloudflare D1 (مجاني)
- النشر: GitHub Actions → Wrangler CLI
- SSH: عبر GitHub Actions (وليس داخل Cloudflare Worker مباشرة)
- الذكاء الاصطناعي: OpenRouter
- المصادقة: JWT يدوي

ملاحظة مهمة: المستودع مُنسوخ بـ shallow clone، فقط 3 ملفات موجودة فعليًا على القرص
(README.md، docs/architecture.md، executor/README.md). لكن الملفات المتبالة موجودة بالكامل
في git tree ويمكن قراءتها عبر `git show HEAD:filename`.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚠️ اختلاف رئيسي: handoff.txt غير موجود في المستودع

لم أجده في:
- الملفات الموجودة على القرص ❌
- ملفات git المُتتبعة (git ls-tree) ❌
- سجل الـ commit الوحيد ❌

المستخدم ذكر أن لديه "نسخة مرجعية خاصة من handoff.txt" وأن المعلومات في رسالته هي المرجع
الحالي.

⚠️ اختلاف طفيف: لا يوجد package.json في الجذر

المشروع الرئيسي لا يحتوي على package.json في الجذر - هو مشروع Cloudflare Pages بحت
(Frontend HTML + Cloudflare Functions). يوجد package.json فقط في executor/ للخدمة المنفصلة.
هذا يعني أن npm install / npm run build لا تنطبق مباشرة على المشروع الرئيسي.

⚠️ deploy.yml موجود بالفعل

المرجع يذكر أن الخطوة التالية هي إضافة أوامر بناء/نشر، لكن ملف deploy.yml موجود بالفعل
وهو يعمل تلقائيًا عند push to main.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
7. أخطاء أو مشاكل واضحة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. لا يوجد handoff.txt في المستودع - قد يكون المستخدم يحتفظ به في مكان آخر.

2. الملفات غير موجودة على القرص (بسبب shallow clone) - لكن هذا لا يؤثر على النشر
   لأن GitHub Actions يعمل checkout كامل.

3. لا يوجد package.json في الجذر ولا node_modules/ - المشروع لا يحتاج npm install
   في Cloudflare Pages.

4. JWT code duplication - دالة verifyJWT مُكررة في عدة ملفات (كل ملف API يعيد
   تعريفها من الصفر بدل import مشترك). مشكلة صيانة لكن ليست خطأ.

5. executor/server.js غير مستخدم حاليًا - يبدو أنه خيار بديل للـ SSH لم يُستخدم بعد
   (SSH يمر عبر GitHub Actions بدلاً منه).

6. لا يوجد tsconfig.json أو package.json في الجذر - لا يوجد TypeScript ولا Node.js
   dependencies للمشروع الرئيسي.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
8. هل توجد بنية مناسبة لـ npm install / npm run build؟
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

لا بالشكل التقليدي.

المشروع هو:
- Cloudflare Pages (Frontend = ملفات HTML جاهزة)
- Cloudflare Functions (Backend = JS files بدون build)
- D1 (قاعدة بيانات SQL)

build process الحالي في deploy.yml هو:
  rm -rf dist
  mkdir -p dist
  cp -r frontend/. dist/
  cp -r functions dist/

هذا ليس build step تقليديًا - هو فقط نسخ ملفات. لا يوجد:
- TypeScript → JavaScript compilation
- Bundling (Vite/Webpack)
- Tree shaking
- Minification

الملف الوحيد الذي يحتاج npm install هو executor/package.json (Express + ssh2).

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
9. الملفات التي ستحتاج غالبًا إلى تعديل في الخطوة التالية
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

إذا كانت الخطوة التالية هي إضافة build step تقليدي (مثلاً تحويل Frontend إلى React/Vite):

1. package.json (جديد في الجذر) - لإضافة dependencies و build scripts
2. deploy.yml - لتعديل build step
3. wrangler.toml - قد يحتاج تعديل pages_build_output_dir
4. frontend/* - قد تحتاج تحويل إلى React components

إذا كان الهدف فقط توثيق / إضافة Scripts:

1. package.json (جديد)
2. deploy.yml (تعديل بسيط)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
10. اقتراحي للخطوة التالية
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

بناءً على ما وجدته، هناك احتمالان:

الاحتمال الأول (الأرجح): الخطوة التالية في المرجع ("npm install / npm run build") قد لا
تنطبق بشكل مباشر لأن المشروع لا يحتاج build step تقليدي. الـ deploy workflow موجود
بالفعل ويعمل.

الاحتمال الثاني: إذا كان المستخدم ينوي تحويل المشروع إلى بنية أقدم (مثلاً React/Vite
مع TypeScript)، فهذه خطوة كبيرة وتحتاج خطوات متعددة.

توصيتي: السؤال المستخدم عن:
1. هل يوجد handoff.txt في مكان آخر يحتاج إلى إنشائه في المستودع؟
2. ما المقصود بـ "npm install / npm run build" بالتحديد؟

لأن المشروع حاليًا يعمل كـ Cloudflare Pages بحت بدون أي build step، وإضافة npm install /
npm run build تتطلب تحديدًا ما المطلوب إضافته.

القواعد الثابتة للمشروع (من المرجع)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. لا يوجد shell حر أو صلاحيات مفتوحة.
2. التنفيذ الفعلي يجب أن يظل عبر مسارات مقيدة وآمنة.
3. الـ AI يخطط أولًا، والتنفيذ لا يحدث إلا بعد موافقة المستخدم.
4. الأسرار لا توضع في Git.
5. لا تعرض أو تطبع أي secrets (SSH password، API keys، Tokens، إلخ).
6. لا تجعل أي Secret يصل إلى المتصفح.
7. لا تعمل أي تعديل خارج نطاق المهمة الحالية.
8. لا تعمل Refactoring جانبي.
9. لا تغير Architecture بدون طلب واضح.
10. المشروع يجب أن يظل قابلًا للتسليم والاختبار من الموبايل.
11. الخطوات تكون صغيرة وقابلة للاختبار.
12. بعد تنفيذ أي خطوة مستقبلية، يجب انتظار تأكيد المستخدم.
13. handoff.txt لا يتم تحديثه إلا بعد تأكيد المستخدم الصريح.
